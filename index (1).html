String(v) : ''; }; const data = rows.map(r => ({ mes: r[idx('Mês de referência')], carimbo: formatCarimbo(r[idx('Carimbo de data/hora')]), parcelaSpim: Number(r[idx('Parcela Spim (R$)')]) || 0, combustivelSpim: Number(r[idx('Combustível Spim (R$)')]) || 0, concertosSpim: Number(r[idx('Concertos/Manutenção Spim (R$)')]) || 0, ipva: Number(r[idx('IPVA (R$)')]) || 0, internet: Number(r[idx('Internet (R$)')]) || 0, luz: Number(r[idx('Luz (R$)')]) || 0, agua: Number(r[idx('Água (R$)')]) || 0, padaria: Number(r[idx('Padaria (R$)')]) || 0, leite: Number(r[idx('Leite (R$)')]) || 0, abasce: Number(r[idx('Abasce (R$)')]) || 0, limpeza: Number(r[idx('Limpeza do templo (R$)')]) || 0, planoCooperativo: Number(r[idx('Plano cooperativo (R$)')]) || 0, combustivelPastor: Number(r[idx('Combustível Pastor (R$)')]) || 0, combustivelIrMarcos: Number(r[idx('Combustível Ir.
